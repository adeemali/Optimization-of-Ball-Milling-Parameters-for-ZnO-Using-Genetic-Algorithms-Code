import random

# ============================================================
# SETTINGS

N_TREES = 200
POP_SIZE = 40
N_GEN = 30
LAMBDA_SIZE = 60
UNCERTAINTY_WEIGHT = 0.5

# ============================================================

# ============================================================

data = [
    [400, 2, 100, 1, None, 0.1, "Zirconia", "Dry", None, 9.51],
    [250, 1200, 30, 10, 30, None, "Tungsten carbide", "Wet", 69.43, 13.07],
    [300, 2, 15, 15, 5, 10, "Steel", "Dry", None, 22],
    [300, 15, 20, 6, None, None, "Stainless-steel", "Dry", 240, 26],
    [300, 2, 10, 20, 15, None, "Chromium steel", "Dry", None, 15],
    [100, 5, None, 1, None, 5, "Zirconia", "Dry", 25, 22],
    [300, 3000, 100, 2, None, None, "Stainless Steel", "Dry", None, 15]
]

columns = [
    "Rotation_speed",
    "Milling_time",
    "Ball_to_powder_ratio",
    "Ball_diameter",
    "Number_of_balls",
    "Precursor_amount",
    "Ball_material",
    "Milling_atmosphere",
    "Initial_crystallite_size",
    "Final_crystallite_size"
]

df = pd.DataFrame(data, columns=columns)

# ============================================================
# PREPROCESSING
# ============================================================

})

    "Wet": 1

df_clean = df.drop(columns=[

    n_estimators=N_TREES,
    random_state=42,
    max_depth=None,
    min_samples_leaf=1,
    bootstrap=False,
    n_jobs=-1

# ============================================================
# UNCERTAINTY FUNCTION
# ============================================================

def predict_with_uncertainty(input_scaled):
    tree_preds = np.array([
        tree.predict(input_scaled)[0]
        for tree in model.estimators_
    ])

    mean_pred = tree_preds.mean()
    std_pred = tree_preds.std()

    return mean_pred, std_pred

# ============================================================
# GA SETUP
# ============================================================

try:
    del creator.FitnessMin
    del creator.Individual
except:
    pass

creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
creator.create("Individual", list, fitness=creator.FitnessMin)

toolbox = base.Toolbox()

bounds = [
    (100, 600),
    (1, 3000),
    (10, 100),
    (1, 20),
    (1, 50),
    (0.01, 10),
    (0, int(df_clean["Ball_material_encoded"].max())),
    (0, 1)
]

param_names = [
    "Rotation Speed",
    "Milling Time",
    "Ball:Powder Ratio",
    "Ball Diameter",
    "Number of Balls",
    "Precursor Amount",
    "Ball Material Encoded",
    "Milling Atmosphere"
]

def create_individual():
    return creator.Individual([
        random.uniform(low, high) for low, high in bounds
    ])

toolbox.register("individual", create_individual)
toolbox.register("population", tools.initRepeat, list, toolbox.individual)

def repair_individual(individual):
    repaired = []

    for i, value in enumerate(individual):
        low, high = bounds[i]
        value = np.clip(value, low, high)

        if i in [4, 6, 7]:
            value = int(round(value))

        repaired.append(value)

    return repaired

def evaluate_standard(individual):
    repaired = repair_individual(individual)
    scaled = scaler.transform([repaired])
    pred = model.predict(scaled)[0]
    return (pred,)

def evaluate_uncertainty(individual):
    repaired = repair_individual(individual)
    scaled = scaler.transform([repaired])

    mean_pred, std_pred = predict_with_uncertainty(scaled)

    # Lower confidence bound objective
    # mean - uncertainty encourages exploration
    objective = mean_pred - UNCERTAINTY_WEIGHT * std_pred

    return (objective,)

toolbox.register("evaluate", evaluate_standard)
toolbox.register("mate", tools.cxBlend, alpha=0.4)
toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.25, indpb=0.2)
toolbox.register("select", tools.selTournament, tournsize=3)

# ============================================================
# GA RUNNER
# ============================================================

def run_ga_variant(
    variant_name,
    uncertainty=False,
    population_size=POP_SIZE,
    generations=N_GEN,
    cxpb=0.6,
    mutpb=0.3,
    lambda_=LAMBDA_SIZE,
    mu=POP_SIZE
):
    if uncertainty:
        toolbox.unregister("evaluate")
        toolbox.register("evaluate", evaluate_uncertainty)
    else:
        toolbox.unregister("evaluate")
        toolbox.register("evaluate", evaluate_standard)

    pop = toolbox.population(n=population_size)
    hof = tools.HallOfFame(1)

    convergence = []
    parameter_evolution = []

    for ind in pop:
        ind.fitness.values = toolbox.evaluate(ind)

    hof.update(pop)

    for gen in range(generations):

        if variant_name in ["ManualGA", "eaSimple"]:
            offspring = algorithms.varAnd(pop, toolbox, cxpb=cxpb, mutpb=mutpb)

            for ind in offspring:
                ind.fitness.values = toolbox.evaluate(ind)

            pop = toolbox.select(offspring, k=population_size)

        elif variant_name == "eaMuPlusLambda":
            offspring = algorithms.varOr(pop, toolbox, lambda_=lambda_, cxpb=cxpb, mutpb=mutpb)

            for ind in offspring:
                ind.fitness.values = toolbox.evaluate(ind)

            pop = toolbox.select(pop + offspring, k=mu)

        elif variant_name == "eaMuCommaLambda":
            offspring = algorithms.varOr(pop, toolbox, lambda_=lambda_, cxpb=cxpb, mutpb=mutpb)

            for ind in offspring:
                ind.fitness.values = toolbox.evaluate(ind)

            pop = toolbox.select(offspring, k=mu)

        elif variant_name == "eaGenerateUpdate":
            offspring = algorithms.varOr(pop, toolbox, lambda_=lambda_, cxpb=cxpb, mutpb=mutpb)

            for ind in offspring:
                ind.fitness.values = toolbox.evaluate(ind)

            pop = toolbox.select(pop + offspring, k=population_size)

        elif variant_name == "eaGenerateUpdate2":
            offspring = algorithms.varOr(pop, toolbox, lambda_=lambda_, cxpb=0.7, mutpb=0.25)

            for ind in offspring:
                ind.fitness.values = toolbox.evaluate(ind)

            pop = toolbox.select(pop + offspring, k=population_size)

        hof.update(pop)

        best_repaired = repair_individual(hof[0])
        best_scaled = scaler.transform([best_repaired])
        true_mean = model.predict(best_scaled)[0]
        _, true_std = predict_with_uncertainty(best_scaled)

        convergence.append(true_mean)
        parameter_evolution.append(best_repaired)

    return hof[0], convergence, np.array(parameter_evolution)

# ============================================================
# RUN STANDARD GA VARIANTS
# ============================================================

random.seed(42)
np.random.seed(42)

ga_variants = [
    "ManualGA",
    "eaSimple",
    "eaMuPlusLambda",
    "eaMuCommaLambda",
    "eaGenerateUpdate",
    "eaGenerateUpdate2"
]

ga_results = {}
ga_convergence = {}
ga_param_evolution = {}

for variant in ga_variants:
    best_ind, convergence, param_evolution = run_ga_variant(
        variant_name=variant,
        uncertainty=False
    )

    ga_results[variant] = best_ind
    ga_convergence[variant] = convergence
    ga_param_evolution[variant] = param_evolution

# ============================================================
# RUN UNCERTAINTY-AWARE GA
# ============================================================

uncertain_results = {}
uncertain_convergence = {}

for variant in ga_variants:
    best_ind, convergence, param_evolution = run_ga_variant(
        variant_name=variant,
        uncertainty=True
    )

    uncertain_results[variant] = best_ind
    uncertain_convergence[variant] = convergence

# ============================================================
# FORMAT GA RESULTS
# ============================================================

def format_results(result_dict, label_prefix):
    rows = []

    for variant, individual in result_dict.items():
        repaired = repair_individual(individual)
        scaled = scaler.transform([repaired])

        mean_pred, std_pred = predict_with_uncertainty(scaled)

        material_code = repaired[6]
        material_name = material_mapping.get(material_code, "Unknown")
        atmosphere = "Wet" if repaired[7] == 1 else "Dry"

        rows.append([
            label_prefix + " - " + variant,
            repaired[0],
            repaired[1],
            repaired[2],
            repaired[3],
            repaired[4],
            repaired[5],
            material_code,
            material_name,
            atmosphere,
            mean_pred,
            std_pred
        ])

    return rows

all_rows = []
all_rows.extend(format_results(ga_results, "Standard GA"))
all_rows.extend(format_results(uncertain_results, "Uncertainty-Aware GA"))

results_df = pd.DataFrame(all_rows, columns=[
    "Method",
    "Rotation Speed",
    "Milling Time",
    "Ball:Powder Ratio",
    "Ball Diameter",
    "Number of Balls",
    "Precursor Amount",
    "Ball Material Encoded",
    "Ball Material",
    "Milling Atmosphere",
    "Predicted Mean Size",
    "Prediction Uncertainty"
])

results_df = results_df.sort_values(
    by="Predicted Mean Size",
    ascending=True
)

print("\nGA and Uncertainty-Aware GA Results")
print(results_df)

# ============================================================
# BAYESIAN OPTIMIZATION
# ============================================================

search_space = [
    Real(100, 600, name="Rotation_speed"),
    Real(1, 3000, name="Milling_time"),
    Real(10, 100, name="Ball_to_powder_ratio"),
    Real(1, 20, name="Ball_diameter"),
    Integer(1, 50, name="Number_of_balls"),
    Real(0.01, 10, name="Precursor_amount"),
    Integer(0, int(df_clean["Ball_material_encoded"].max()), name="Ball_material_encoded"),
    Integer(0, 1, name="Milling_atmosphere_encoded")
]

def bayes_objective(params):
    repaired = repair_individual(params)
    scaled = scaler.transform([repaired])
    pred = model.predict(scaled)[0]
    return pred

bayes_result = gp_minimize(
    func=bayes_objective,
    dimensions=search_space,
    n_calls=40,
    n_initial_points=10,
    random_state=42,
    acq_func="EI"
)

bayes_best = repair_individual(bayes_result.x)
bayes_scaled = scaler.transform([bayes_best])
bayes_mean, bayes_std = predict_with_uncertainty(bayes_scaled)

bayes_material = material_mapping.get(bayes_best[6], "Unknown")
bayes_atmosphere = "Wet" if bayes_best[7] == 1 else "Dry"

print("\nBayesian Optimization Result")
